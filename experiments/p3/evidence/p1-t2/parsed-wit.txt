package wafer:pipeline@0.2.0;

interface types {
  record lineage {
    parent-id: option<string>,
    trace-id: option<string>,
  }

  record envelope {
    id: string,
    timestamp: u64,
    source: string,
    content-type: string,
    metadata: list<tuple<string, string>>,
    lineage: lineage,
    retry-count: u32,
    payload: list<u8>,
  }

  variant process-error {
    bad-input(string),
    dependency-failed(string),
    processing-failed(string),
    timed-out,
    unrecoverable(string),
  }
}

interface message-transform {
  use types.{envelope, process-error};

  /// Consumes one owned envelope and resolves to one owned envelope or typed error.
  process: async func(input: envelope) -> result<envelope, process-error>;
}

interface stream-transform {
  use types.{envelope, process-error};

  /// Starts one finite session. Closing the input ends production; the future reports
  /// terminal success or a session-level error after the output stream is drained.
  process: func(input: stream<envelope>) -> tuple<stream<result<envelope, process-error>>, future<result<_, process-error>>>;
}

world transform-message-node {
  import types;

  export message-transform;
}
world transform-stream-node {
  import types;

  export stream-transform;
}
